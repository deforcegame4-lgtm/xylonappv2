async function downloadTikTok(tiktokUrl){ const apiUrl = "https://api-faa.my.id/faa/tiktok?url=" + encodeURIComponent(tiktokUrl); const response = await fetch(apiUrl); if(!response.ok){ throw new Error( `Faa API error: ${response.status}` ); } const data = await response.json(); if(data.status !== true){ throw new Error( "Faa API tidak berhasil memproses video." ); } const videoUrl = data.result?.alternatives?.hd || data.result?.alternatives?.selected || data.result?.data; if(!videoUrl){ throw new Error( "URL video tidak ditemukan." ); } window.open( videoUrl, "_blank", "noopener,noreferrer" ); }
